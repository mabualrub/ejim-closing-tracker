Direct Google Apps Script (One-Click Drive Creator)

If you prefer creating the spreadsheet directly inside your Google Drive without uploading a file, copy and run this script.

// Google Apps Script: Creates "EJIM Closing Ceremony Master Tracker" with 9 Worksheets
function createEJIMTrackingSheet() {
  var ss = SpreadsheetApp.create("EJIM Closing Ceremony Master Task & Item Tracker");
  var sheetsData = [
    { name: "Master Schedule", color: "#1e293b" },
    { name: "1. Branding & Identity", color: "#0284c7" },
    { name: "2. Invitations & Copy", color: "#d97706" },
    { name: "3. Program & Media", color: "#059669" },
    { name: "4. Production & Shaaban", color: "#7c3aed" },
    { name: "5. Receiving & Inventory", color: "#ea580c" },
    { name: "6. Venue & AV Engineering", color: "#0891b2" },
    { name: "7. Registration Desk", color: "#db2777" },
    { name: "8. Close-out & Reporting", color: "#475569" }
  ];

  var defaultSheet = ss.getActiveSheet();
  defaultSheet.setName(sheetsData[0].name);
  defaultSheet.setTabColor(sheetsData[0].color);

  for (var i = 1; i < sheetsData.length; i++) {
    var sh = ss.insertSheet(sheetsData[i].name);
    sh.setTabColor(sheetsData[i].color);
  }

  Logger.log("Created successfully at: " + ss.getUrl());
}

How to use in 30 seconds:

  1. Visit script.new in your browser.
  2. Paste the code above into the script editor.
  3. Click Run (grant standard Drive permission once). The new multi-sheet document will appear right in your Google Drive!
